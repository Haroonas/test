#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080

int main() {
    int server_fd, client_fd;
    char buffer[100], reply[20];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    int a, b, c, d;
    char extra;

    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);

    bind(server_fd, (struct sockaddr *)&server, sizeof(server));
    listen(server_fd, 1);

    printf("IP validator server waiting...\n");

    client_fd = accept(server_fd, (struct sockaddr *)&client, &len);

    printf("\nClient connected: %s:%d\n",
           inet_ntoa(client.sin_addr), ntohs(client.sin_port));

    int n = recv(client_fd, buffer, sizeof(buffer) - 1, 0);
    buffer[n] = '\0';

    printf("Received from client: %s\n", buffer);

    // Split at dots into 4 numbers
    int count = sscanf(buffer, "%d.%d.%d.%d%c",
                       &a, &b, &c, &d, &extra);

    if (count == 4 &&
        a >= 0 && a <= 255 &&
        b >= 0 && b <= 255 &&
        c >= 0 && c <= 255 &&
        d >= 0 && d <= 255)
        strcpy(reply, "Valid IP");
    else
        strcpy(reply, "Invalid IP");

    printf("Validation result: %s\n", reply);

    send(client_fd, reply, strlen(reply), 0);

    printf("Result sent to client: %s\n", reply);
    printf("Client request completed.\n");
    printf("Server shutting down...\n");

    close(client_fd);
    close(server_fd);
    return 0;
}
