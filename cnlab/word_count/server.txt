#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080

int main() {
    int server_fd, client_fd, n;
    char buffer[200], reply[100];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);
    int words = 0, chars = 0, upper = 0;

    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR,
               &opt, sizeof(opt));

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);

    bind(server_fd, (struct sockaddr *)&server, sizeof(server));
    listen(server_fd, 1);

    printf("Sentence statistics server waiting...\n");

    client_fd = accept(server_fd, (struct sockaddr *)&client, &len);

    printf("\nClient connected: %s:%d\n",
           inet_ntoa(client.sin_addr), ntohs(client.sin_port));

    n = recv(client_fd, buffer, sizeof(buffer) - 1, 0);
    buffer[n] = '\0';

    printf("Received from client: %s\n", buffer);

    chars = strlen(buffer);

    for (int i = 0; i < chars; i++) {
        if (isupper(buffer[i]))
            upper++;

        if (!isspace(buffer[i]) &&
            (i == 0 || isspace(buffer[i - 1])))
            words++;
    }

    sprintf(reply, "Words: %d, Characters: %d, Uppercase letters: %d",
            words, chars, upper);

    printf("Calculated statistics:\n");
    printf("Words: %d\n", words);
    printf("Characters: %d\n", chars);
    printf("Uppercase letters: %d\n", upper);

    send(client_fd, reply, strlen(reply), 0);

    printf("Result sent to client: %s\n", reply);
    printf("Client request completed.\n");
    printf("Server shutting down...\n");

    close(client_fd);
    close(server_fd);

    return 0;
}
