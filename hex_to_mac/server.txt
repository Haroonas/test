
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080

int main() {
    int sockfd;
    char buffer[100], mac[50];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);
    bind(sockfd, (struct sockaddr *)&server, sizeof(server));

    printf("MAC formatter server waiting...\n");

    int n = recvfrom(sockfd, buffer, sizeof(buffer) - 1, 0,
                     (struct sockaddr *)&client, &len);
    buffer[n] = '\0';

    printf("\nClient connected: %s:%d\n",
           inet_ntoa(client.sin_addr), ntohs(client.sin_port));
    printf("Received from client: %s\n", buffer);

    if (strlen(buffer) != 12) {
        strcpy(mac, "Error: need exactly 12 hex digits");
    } else {
        int j = 0;
        for (int i = 0; i < 12; i++) {
            mac[j++] = toupper(buffer[i]);
            if (i % 2 == 1 && i != 11)
                mac[j++] = ':';
        }
        mac[j] = '\0';
    }

    printf("Formatted MAC address: %s\n", mac);

    sendto(sockfd, mac, strlen(mac), 0,
           (struct sockaddr *)&client, len);

    printf("Formatted MAC address sent to client.\n");
    printf("Client request completed.\n");
    printf("Server shutting down...\n");

    close(sockfd);
    return 0;
}
