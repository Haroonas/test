include <stdio.h>
#include <string.h>
#include <time.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080

int main() {
    int sockfd;
    char buffer[100], reply[100];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);

    sockfd = socket(AF_INET, SOCK_DGRAM, 0);

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);
    bind(sockfd, (struct sockaddr *)&server, sizeof(server));

    printf("Time server waiting...\n");

    int n = recvfrom(sockfd, buffer, sizeof(buffer) - 1, 0,
                     (struct sockaddr *)&client, &len);
    buffer[n] = '\0';

    printf("\nClient connected: %s:%d\n",
           inet_ntoa(client.sin_addr), ntohs(client.sin_port));
    printf("Received from client: %s\n", buffer);

    time_t now = time(NULL);
    struct tm *t = localtime(&now);

    if (strcmp(buffer, "GET_TIME") == 0)
        strftime(reply, sizeof(reply), "Time: %H:%M:%S", t);

    else if (strcmp(buffer, "GET_DATE") == 0)
        strftime(reply, sizeof(reply), "Date: %d-%m-%Y", t);

    else
        strcpy(reply, "Error: use GET_TIME or GET_DATE");

    printf("Generated response: %s\n", reply);

    sendto(sockfd, reply, strlen(reply), 0,
           (struct sockaddr *)&client, len);

    printf("Response sent to client.\n");
    printf("Client request completed.\n");
    printf("Server shutting down...\n");

    close(sockfd);
    return 0;
}
