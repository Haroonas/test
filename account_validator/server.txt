#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

#define PORT 8080

int main() {
    int server_fd, client_fd, n;
    char buffer[100], reply[30];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);

    // Hardcoded accounts
    char *accounts[3] = {
        "admin admin123",
        "alice alice1",
        "bob bob456"
    };
    int found = 0;

    server_fd = socket(AF_INET, SOCK_STREAM, 0);

    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR,
               &opt, sizeof(opt));

    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(PORT);

    bind(server_fd, (struct sockaddr *)&server, sizeof(server));
    listen(server_fd, 1);

    printf("Account validator server waiting...\n");

    client_fd = accept(server_fd,
                       (struct sockaddr *)&client, &len);

    printf("\nClient connected: %s:%d\n",
           inet_ntoa(client.sin_addr), ntohs(client.sin_port));

    n = recv(client_fd, buffer, sizeof(buffer) - 1, 0);
    buffer[n] = '\0';

    printf("Received from client: %s\n", buffer);

    // Compare received credentials with stored accounts
    for (int i = 0; i < 3; i++) {
        if (strcmp(buffer, accounts[i]) == 0)
            found = 1;
    }

    if (found == 1)
        strcpy(reply, "Access Granted");
    else
        strcpy(reply, "Access Denied");

    printf("Validation result: %s\n", reply);

    send(client_fd, reply, strlen(reply), 0);

    printf("Result sent to client: %s\n", reply);
    printf("Client request completed.\n");
    printf("Server shutting down...\n");

    close(client_fd);
    close(server_fd);

    return 0;
}
